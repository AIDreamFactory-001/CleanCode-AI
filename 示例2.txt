    let lockId: string | null = null;
    let productIdSnapshot: string = '';
    let quantitySnapshot: number = 0;
    let userIdSnapshot: string = '';
    let idempotentKey: string = '';

    ctx.temp = ctx.temp || {
      success: true,
      fail_reason: null,
      logInfo: null,
      loginUserId: null
    };
    ctx.shared = ctx.shared || {};
    ctx.deadline = Date.now() + REQUEST_TIMEOUT;
    ctx.requestId = ctx.requestId || `req_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;

          code: ERROR_CODE.PARAM_INVALID,
          msg: '登录已过期，请重新登录',
          data: null,
          detail: null
        });
        return;
      }

      const tokenData = tokenResult.data as Record<string, unknown>;
      if (tokenData.userId !== userId) {
        res.status(403).json({
          code: ERROR_CODE.PARAM_INVALID,
          msg: '身份验证失败',
          data: null,
          detail: null
        });
        return;
      }

      // 幂等检查：用户+商品+当前分钟为幂等键，防止重复下单
      const currentMinute = Math.floor(Date.now() / 60000);
      idempotentKey = `seckill:${userId}:${productId}:${currentMinute}`;
      ctx.shared.idempotentKey = idempotentKey;

      const idemResult = await IdempotentCheck(idempotentKey, 3600, true);
      if (!idemResult.success) {
        res.status(500).json({
          code: ERROR_CODE.INTERNAL_ERROR,
          msg: '服务器繁忙，请稍后再试',

          detail: null

        ctx.temp.success = false;
        ctx.temp.fail_reason = txResult.error;

          { transaction }
        );

      if (!stockDeductResult.success || stockDeductResult.data?.affectedRows === 0) {
        ctx.temp.success = false;
        ctx.temp.fail_reason = '库存不足或扣减失败';

            await LockRelease(lockKey, lockId);

        detail: null
      });

// ===== 标准函数: DbQuery =====
async function DbQuery(tableName: string, conditions: Record<string, any>, options?: { transaction?: Transaction; lock?: string }): Promise<StandardResult<any[]>> {
  if (!tableName || typeof tableName !== 'string') {
    return { success: false, error: 'DbQuery: 表名不能为空' };
  }
  let connection: PoolConnection | null = null;
  try {
    if (options?.transaction) {
      connection = options.transaction.connection;
    } else {
      connection = await mysqlPool.getConnection();
    }
    const keys = Object.keys(conditions || {});
    const whereClause = keys.length ? `WHERE ${keys.map(k => `\`${k}\` = ?`).join(' AND ')}` : '';
    const lockClause = options?.lock ? ` ${options.lock}` : '';
    const values = keys.map(k => conditions[k]);
    const [rows] = await connection.query(`SELECT * FROM \`${tableName}\` ${whereClause}${lockClause}`, values);
    if (!options?.transaction && connection) {
      connection.release();
    }
    return { success: true, data: rows as any[] };
  } catch (err) {
    if (connection && !options?.transaction) {
      try { connection.release(); } catch {}
    }
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: DbExecute =====
async function DbExecute(tableName: string, operation: 'insert' | 'update' | 'delete', data: Record<string, any>, conditions?: Record<string, any>, options?: { transaction?: Transaction }): Promise<StandardResult<ResultSetHeader>> {
  if (!tableName || !operation || !['insert', 'update', 'delete'].includes(operation)) {
    return { success: false, error: 'DbExecute: 无效参数' };
  }
  if (operation !== 'insert' && !data) {
    return { success: false, error: 'DbExecute: 操作数据不能为空' };
  }
  let connection: PoolConnection | null = null;
  try {
    if (options?.transaction) {
      connection = options.transaction.connection;
    } else {
      connection = await mysqlPool.getConnection();
    }
    let sql: string;
    let values: any[];
    if (operation === 'insert') {
      const keys = Object.keys(data);
      const placeholders = keys.map(k => {
        if (typeof data[k] === 'function') {
          return data[k]();
        }
        return '?';
      });
      sql = `INSERT INTO \`${tableName}\` (${keys.map(k => `\`${k}\``).join(', ')}) VALUES (${placeholders.join(', ')})`;
      values = keys.filter(k => typeof data[k] !== 'function').map(k => data[k]);
    } else if (operation === 'update') {
      const setKeys = Object.keys(data);
      const whereKeys = Object.keys(conditions || {});
      const setClause = setKeys.map(k => {
        if (typeof data[k] === 'function') {
          return `\`${k}\` = ${data[k]()}`;
        }
        return `\`${k}\` = ?`;
      }).join(', ');
      const whereClause = whereKeys.length ? `WHERE ${whereKeys.map(k => `\`${k}\` = ?`).join(' AND ')}` : '';
      sql = `UPDATE \`${tableName}\` SET ${setClause} ${whereClause}`;
      values = [
        ...setKeys.filter(k => typeof data[k] !== 'function').map(k => data[k]),
        ...whereKeys.map(k => (conditions as Record<string, any>)[k])
      ];
    } else { // delete
      const whereKeys = Object.keys(conditions || {});
      const whereClause = whereKeys.length ? `WHERE ${whereKeys.map(k => `\`${k}\` = ?`).join(' AND ')}` : '';
      sql = `DELETE FROM \`${tableName}\` ${whereClause}`;
      values = whereKeys.map(k => (conditions as Record<string, any>)[k]);
    }
    const [result] = await connection.query(sql, values);
    if (!options?.transaction && connection) {
      connection.release();
    }
    return { success: true, data: result as ResultSetHeader };
  } catch (err) {
    if (connection && !options?.transaction) {
      try { connection.release(); } catch {}
    }
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: BeginTransaction =====
async function BeginTransaction(): Promise<StandardResult<Transaction>> {
  try {
    const connection = await mysqlPool.getConnection();
    await connection.beginTransaction();
    const transactionId = `tx_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    return { success: true, data: { id: transactionId, connection } };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: CommitTransaction =====
async function CommitTransaction(transaction: Transaction): Promise<StandardResult> {
  if (!transaction || !transaction.id || !transaction.connection) {
    return { success: false, error: 'CommitTransaction: 无效事务对象' };
  }
  try {
    await transaction.connection.commit();
    transaction.connection.release();
    return { success: true };
  } catch (err) {
    try { 
      await transaction.connection.rollback();
      transaction.connection.release();
    } catch {}
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: RollbackTransaction =====
async function RollbackTransaction(transaction: Transaction): Promise<StandardResult> {
  if (!transaction || !transaction.id || !transaction.connection) {
    return { success: false, error: 'RollbackTransaction: 无效事务对象' };
  }
  try {
    await transaction.connection.rollback();
    transaction.connection.release();
    return { success: true };
  } catch (err) {
    try { transaction.connection.release(); } catch {}
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: CompensateBind =====
async function CompensateBind(ctx: Express.Request['ctx'], opId: string, rollback: () => Promise<void>): Promise<StandardResult<{ bound: boolean; stackSize: number }>> {
  if (!ctx || typeof ctx !== 'object' || typeof rollback !== 'function') {
    return { success: false, error: 'CompensateBind: 无效参数' };
  }
  if (!ctx.shared) ctx.shared = {};
  if (!ctx.shared.compensateStack) ctx.shared.compensateStack = [];
  ctx.shared.compensateStack.push({
    id: opId || `comp_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`,
    rollback: rollback,
    executed: false,
    boundAt: Date.now()
  });
  return { success: true, data: { bound: true, stackSize: ctx.shared.compensateStack.length } };
}

// ===== 标准函数: CompensateRun =====
async function CompensateRun(ctx: Express.Request['ctx']): Promise<StandardResult> {
  if (!ctx || typeof ctx !== 'object') {
    return { success: false, error: 'CompensateRun: 无效上下文' };
  }
  if (!ctx.shared?.compensateStack || !Array.isArray(ctx.shared.compensateStack) || ctx.shared.compensateStack.length === 0) {
    return { success: true, data: { executed: 0 } };
  }
  const stack = [...ctx.shared.compensateStack].reverse(); // 逆序执行补偿
  let executed = 0;
  for (const item of stack) {
    if (item.executed) continue;
    try {
      await item.rollback();
      item.executed = true;
      executed++;
    } catch (err) {
      return { success: false, error: `补偿操作 ${item.id} 失败: ${(err as Error).message}` };
    }
  }
  ctx.shared.compensateStack = [];
  return { success: true, data: { executed } };
}

// ===== 标准函数: Retry =====
async function Retry<T>(fn: () => Promise<T>, options: { maxRetries: number; backoffFactor: number; initialDelay: number }): Promise<StandardResult<T>> {
  if (typeof fn !== 'function' || options.maxRetries < 0) {
    return { success: false, error: 'Retry: 无效参数' };
  }
  let attempt = 0;
  let lastError: Error | null = null;
  while (attempt <= options.maxRetries) {
    try {
      const result = await fn();
      return { success: true, data: result };
    } catch (err) {
      lastError = err as Error;
      attempt++;
      if (attempt > options.maxRetries) break;
      const delay = Math.min(options.initialDelay * Math.pow(options.backoffFactor, attempt - 1), 1000);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
  return { success: false, error: lastError?.message || `重试${options.maxRetries}次后仍然失败` };
}

// ===== 标准函数: IdempotentCheck =====
async function IdempotentCheck(idempotentKey: string, ttl: number, autoCache: boolean): Promise<StandardResult<{ is_first: boolean; cachedResponse?: any }>> {
  if (!idempotentKey || typeof idempotentKey !== 'string' || ttl < 0) {
    return { success: false, error: 'IdempotentCheck: 无效参数' };
  }
  try {
    const exists = await redis.get(idempotentKey);
    if (exists) {
      try {
        const cachedData = JSON.parse(exists);
        return { success: true, data: { is_first: false, cachedResponse: cachedData } };
      } catch {
        return { success: true, data: { is_first: false, cachedResponse: { status: 'SUCCESS' } } };
      }
    }
    if (autoCache) {
      await redis.set(idempotentKey, JSON.stringify({ status: 'PROCESSING', confirmed: false }), 'EX', ttl);
    }
    return { success: true, data: { is_first: true } };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: IdempotentConfirm =====
async function IdempotentConfirm(key: string): Promise<StandardResult<{ confirmed: boolean; existed: boolean }>> {
  if (!key || typeof key !== 'string') {
    return { success: false, error: 'IdempotentConfirm: 无效key' };
  }
  const getResult = await CacheGet(key);
  if (!getResult.success) {
    return { success: false, error: getResult.error };
  }
  const item = getResult.data;
  if (!item) {
    return { success: true, data: { confirmed: false, existed: false } };
  }
  if (item.confirmed === true) {
    return { success: true, data: { confirmed: true, existed: true } };
  }
  const value = { ...item, confirmed: true };
  const ttl = await redis.ttl(key);
  const setResult = await CacheSet(key, value, ttl > 0 ? ttl : 3600);
  if (!setResult.success) {
    return { success: false, error: setResult.error };
  }
  return { success: true, data: { confirmed: true, existed: true } };
}

// ===== 标准函数: LockAcquire =====
async function LockAcquire(lockKey: string, expireSeconds: number, retryTimes: number, isBlocking: boolean): Promise<StandardResult<{ locked: boolean; lockId?: string }>> {
  if (!lockKey || typeof lockKey !== 'string' || expireSeconds < 1 || retryTimes < 0) {
    return { success: false, error: 'LockAcquire: 无效参数' };
  }
  const lockId = `lock_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
  let attempts = 0;
  while (attempts <= retryTimes) {
    try {
      const result = await redis.set(lockKey, lockId, 'NX', 'EX', expireSeconds);
      if (result === 'OK') {
        return { success: true, data: { locked: true, lockId } };
      }
      if (!isBlocking) break;
      await new Promise(resolve => setTimeout(resolve, 10));
      attempts++;
    } catch (err) {
      return { success: false, error: (err as Error).message };
    }
  }
  return { success: true, data: { locked: false } };
}

// ===== 标准函数: LockRelease（使用Lua脚本校验lockId防误删） =====
async function LockRelease(lockKey: string, lockId: string): Promise<StandardResult> {
  if (!lockKey || !lockId || typeof lockKey !== 'string' || typeof lockId !== 'string') {
    return { success: false, error: 'LockRelease: 无效参数' };
  }
  try {
    const script = `
      if redis.call('GET', KEYS[1]) == ARGV[1] then
        return redis.call('DEL', KEYS[1])
      else
        return 0
      end
    `;
    await redis.eval(script, 1, lockKey, lockId);
    return { success: true };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: checkTimeout =====
function checkTimeout(ctx: Express.Request['ctx']): boolean {
  if (!ctx.deadline) return false;
  const isTimeout = Date.now() > ctx.deadline;
  if (isTimeout) {
    ctx.temp.success = false;
    ctx.temp.fail_reason = '请求超时';
  }
  return isTimeout;
}

// ===== 标准函数: CacheGet =====
async function CacheGet(key: string): Promise<StandardResult<any>> {
  if (!key || typeof key !== 'string') {
    return { success: false, error: 'CacheGet: 无效key' };
  }
  try {
    const value = await redis.get(key);
    if (!value) return { success: true, data: null };
    try {
      return { success: true, data: JSON.parse(value) };
    } catch {
      return { success: true, data: value };
    }
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: CacheSet =====
async function CacheSet(key: string, value: any, ttl: number | null): Promise<StandardResult> {
  if (!key || typeof key !== 'string') {
    return { success: false, error: 'CacheSet: 无效key' };
  }
  try {
    const serialized = typeof value === 'string' ? value : JSON.stringify(value);
    if (ttl && ttl > 0) {
      await redis.set(key, serialized, 'EX', ttl);
    } else {
      await redis.set(key, serialized);
    }
    return { success: true };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

    return { success: true, data: str };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: ValidateParams =====
async function ValidateParams(params: Record<string, any>, rules: ValidationRules): Promise<StandardResult<{ validated: SeckillRequest; errors: string[] }>> {
  if (!params || !rules || typeof params !== 'object' || typeof rules !== 'object') {
    return { success: false, error: 'ValidateParams: 无效参数' };
  }
  try {
    const errors: string[] = [];
    const validated: Partial<SeckillRequest> = {};
    for (const [key, rule] of Object.entries(rules)) {
      const value = params[key];
      if (rule.required && (value === undefined || value === null || (typeof value === 'string' && value.trim() === ''))) {
        errors.push(`${key}不能为空`);
        continue;
      }
      if (value !== undefined && value !== null) {
        if (rule.type === 'string') {
          if (typeof value !== 'string') {
            errors.push(`${key}必须为字符串`);
            continue;
          }
          const trimmedValue = value.trim();
          if (rule.min !== undefined && trimmedValue.length < rule.min) {
            errors.push(`${key}长度不能少于${rule.min}位`);
          }
          if (rule.max !== undefined && trimmedValue.length > rule.max) {
            errors.push(`${key}长度不能超过${rule.max}位`);
          }
          validated[key as keyof SeckillRequest] = trimmedValue as any;
        } else if (rule.type === 'number') {
          if (typeof value !== 'number' || isNaN(value)) {
            errors.push(`${key}必须为数字`);
            continue;
          }
          if (rule.min !== undefined && value < rule.min) {
            errors.push(`${key}不能小于${rule.min}`);
          }
          if (rule.max !== undefined && value > rule.max) {
            errors.push(`${key}不能大于${rule.max}`);
          }
          if (!Number.isInteger(value)) {
            errors.push(`${key}必须为整数`);
          }
          validated[key as keyof SeckillRequest] = value as any;
        }
      }
    }
    if (errors.length > 0) {
      return { success: false, data: { errors } as any };
    }
    return { success: true, data: { validated: validated as SeckillRequest, errors: [] } };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}
