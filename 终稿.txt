  DEFAULT: 300,

// 路由超时配置（3秒）
const REQUEST_TIMEOUT = 3000;

router.post('/api/v1/user/register',
  async (req: Request, res: Response, next: NextFunction): Promise<void> => {
    // 初始化上下文（强制保障ctx存在）
    req.ctx = req.ctx || {} as any;
    req.ctx.temp = { success: true, fail_reason: null, logInfo: null };
    req.ctx.shared = {};
    req.ctx.deadline = Date.now() + REQUEST_TIMEOUT;
    req.ctx.requestId = req.ctx.requestId || `req_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    req.ctx.ip = req.ip || req.headers['x-forwarded-for'] as string || req.socket.remoteAddress || '0.0.0.0';

    let transaction: Transaction | null = null;
    let lockId: string | null = null;
    const LIGHT = false; // 标准级写入场景，非轻量请求
    let registeredUserId: string | null = null;

    try {
      // ===== Stage 2: 前置校验与合规检查 =====
      // 请求体合法性校验
      const body = req.body as unknown as Record<string, unknown>;
      if (!body || typeof body !== 'object' || Array.isArray(body)) {
        res.status(400).json({
          code: ERROR_CODE.PARAM_INVALID,
          msg: '请求体不能为空且必须为JSON对象',
          data: null,
          detail: null
        });
        return;
      }

      // 结构化参数校验
      const validateResult = await ValidateParams(body, REGISTER_VALIDATION_RULES);
      if (!validateResult.success) {
        const errors = validateResult.data?.errors as string[] || [];
        res.status(400).json({
          code: ERROR_CODE.PARAM_INVALID,
          msg: `参数校验失败: ${errors.join(', ')}`,
          data: null,
          detail: null
        });
        return;
      }
      const validated = validateResult.data?.validated as RegisterRequest;
      const { username, password, email } = validated;

      // 幂等检查
      const idempotentKey = `user:register:${req.ctx.requestId}`;
      req.ctx.shared.idempotentKey = idempotentKey;
      const idemResult = await IdempotentCheck(idempotentKey, CACHE_TTL.LONG, true);
      if (!idemResult.success) {
        await LogError(`幂等检查失败: ${idemResult.error}`, req.ctx.userId, 'user_register', 'ERROR', req.ctx.requestId, req.ctx.ip);
        res.status(500).json({
          code: ERROR_CODE.INTERNAL_ERROR,
          msg: '服务内部错误',
          data: null,
          detail: null
        });
        return;
      }
      if (!idemResult.data?.is_first) {
        const cached = idemResult.data?.cachedResponse || { username, status: 'SUCCESS' };
        res.status(200).json({
          code: ERROR_CODE.SUCCESS,
          msg: '注册请求已处理',
          data: cached,
          detail: null
        });
        return;
      }

      // 超时检查
      if (checkTimeout(req.ctx)) {
        throw new Error('TIMEOUT');
      }

      // ===== Stage 3: 核心业务执行 =====
      // 获取分布式锁（防止并发重复注册）
      const lockKey = `user:register:lock:${username}`;
      const lockResult = await LockAcquire(lockKey, 10, 3, false);
      if (!lockResult.success || !lockResult.data?.locked) {
        res.status(429).json({
          code: ERROR_CODE.LOCK_FAIL,
          msg: '操作过于频繁，请稍后重试',
          data: null,
          detail: null
        });
        return;
      }
      lockId = lockResult.data.lockId;
      req.ctx.shared.lockKey = lockKey;
      req.ctx.shared.lockId = lockId;

      // 超时检查
      if (checkTimeout(req.ctx)) {
        throw new Error('TIMEOUT');
      }

      // 检查用户名是否已存在
      const usernameExistResult = await DbQuery('users', { username });
      if (!usernameExistResult.success) {
        req.ctx.temp.success = false;
        req.ctx.temp.fail_reason = `用户名查询失败: ${usernameExistResult.error}`;
        throw new Error('DB_QUERY_FAILED');
      }
      if (usernameExistResult.data?.length > 0) {
        res.status(409).json({
          code: ERROR_CODE.USER_ALREADY_EXISTS,
          msg: '用户名已存在',
          data: null,
          detail: null
        });
        return;
      }

      // 检查邮箱是否已存在
      const emailExistResult = await DbQuery('users', { email });
      if (!emailExistResult.success) {
        req.ctx.temp.success = false;
        req.ctx.temp.fail_reason = `邮箱查询失败: ${emailExistResult.error}`;
        throw new Error('DB_QUERY_FAILED');
      }
      if (emailExistResult.data?.length > 0) {
        res.status(409).json({
          code: ERROR_CODE.EMAIL_ALREADY_EXISTS,
          msg: '邮箱已存在',
          data: null,
          detail: null
        });
        return;
      }

      // 密码哈希处理
      const passwordHashResult = await Sha256(password);
      if (!passwordHashResult.success) {
        req.ctx.temp.success = false;
        req.ctx.temp.fail_reason = `密码加密失败: ${passwordHashResult.error}`;
        throw new Error('PASSWORD_HASH_FAILED');
      }
      const passwordHash = passwordHashResult.data;

      // 开启事务
      if (checkTimeout(req.ctx)) {
        throw new Error('TIMEOUT');
      }
      const txResult = await BeginTransaction();
      if (!txResult.success) {
        req.ctx.temp.success = false;
        req.ctx.temp.fail_reason = `事务开启失败: ${txResult.error}`;
        throw new Error('TX_BEGIN_FAILED');
      }
      transaction = txResult.data;
      req.ctx.shared.transaction = transaction;

      // 写入用户数据
      const createUserResult = await DbExecute('users', 'insert', {
        username,
        password_hash: passwordHash,
        email,
        status: USER_STATUS.ACTIVE,
        created_at: new Date(),
        updated_at: new Date()
      }, undefined, { transaction });
      if (!createUserResult.success) {
        req.ctx.temp.success = false;
        req.ctx.temp.fail_reason = `用户创建失败: ${createUserResult.error}`;
        throw new Error('USER_CREATE_FAILED');
      }
      registeredUserId = (createUserResult.data as ResultSetHeader).insertId.toString();

      // ===== Stage 4: 后置归集与资源收尾 =====
      // 提交事务
      if (checkTimeout(req.ctx)) {
        throw new Error('TIMEOUT');
      }
      const commitResult = await CommitTransaction(transaction);
      if (!commitResult.success) {
        req.ctx.temp.success = false;
        req.ctx.temp.fail_reason = `事务提交失败: ${commitResult.error}`;
        throw new Error('COMMIT_FAILED');
      }
      transaction = null;

      // 释放锁
      if (lockId) {
        await LockRelease(lockKey, lockId);
        lockId = null;
      }

      // 幂等确认
      const responseData = {
        userId: registeredUserId,
        username,
        email,
        status: USER_STATUS.ACTIVE
      };
      await IdempotentConfirm(idempotentKey, responseData);

      // 记录操作日志
      req.ctx.temp.logInfo = {
        action: 'user_register',
        username,
        email,
        userId: registeredUserId
      };
      const logUserId = req.ctx.userId || registeredUserId;
      await Log(req.ctx.temp.logInfo, logUserId, 'user', 'INFO', req.ctx.requestId, req.ctx.ip);

      // ===== Stage 5: 统一契约响应 =====
      res.status(200).json({
        code: ERROR_CODE.SUCCESS,
        msg: '用户注册成功',
        data: responseData,
        detail: null
      });
    } catch (err) {
      req.ctx.temp.success = false;
      if (!req.ctx.temp.fail_reason) {
        req.ctx.temp.fail_reason = (err as Error).message;
      }

      // 异常资源清理（强制保障资源释放）
      if (!LIGHT) {
        const lockKey = req.ctx.shared.lockKey as string;
        if (transaction) {
          await RollbackTransaction(transaction).catch(e => {
            LogError(`事务回滚异常: ${e.message}`, req.ctx.userId, 'transaction', 'ERROR', req.ctx.requestId, req.ctx.ip);
          });
          transaction = null;
        }
        if (lockId && lockKey) {
          await LockRelease(lockKey, lockId).catch(e => {
            LogError(`锁释放异常: ${e.message}`, req.ctx.userId, 'lock', 'ERROR', req.ctx.requestId, req.ctx.ip);
          });
          lockId = null;
        }
      }

      // 错误日志记录
      const logUserId = req.ctx.userId || req.ctx.temp.loginUserId;
      await LogError(req.ctx.temp.fail_reason, logUserId, 'user_register', 'ERROR', req.ctx.requestId, req.ctx.ip);

      // 错误响应处理
      let errorCode: string = ERROR_CODE.INTERNAL_ERROR;
      let statusCode: number = 500;
      const errMsg = (err as Error).message;

      if (errMsg === 'TIMEOUT') {
        errorCode = ERROR_CODE.TIMEOUT;
        statusCode = 504;
      }

      res.status(statusCode).json({
        code: errorCode,
        msg: '注册失败，请稍后重试',
        data: null,
        detail: null // 生产环境强制脱敏
      });
    }
  }
);

// ===== 标准函数: BeginTransaction =====
async function BeginTransaction(): Promise<StandardResult<Transaction>> {
  try {
    const connection = await mysqlPool.getConnection();
    await connection.beginTransaction();
    const transactionId = `tx_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
    return { success: true, data: { id: transactionId, connection } };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: RollbackTransaction =====
async function RollbackTransaction(transaction: Transaction): Promise<StandardResult> {
  if (!transaction || !transaction.id || !transaction.connection) {
    return { success: false, error: 'RollbackTransaction: invalid transaction object' };
  }
  try {
    await transaction.connection.rollback();
    transaction.connection.release();
    return { success: true };
  } catch (err) {
    try { transaction.connection.release(); } catch {}
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: CommitTransaction =====
async function CommitTransaction(transaction: Transaction): Promise<StandardResult> {
  if (!transaction || !transaction.id || !transaction.connection) {
    return { success: false, error: 'CommitTransaction: invalid transaction object' };
  }
  try {
    await transaction.connection.commit();
    transaction.connection.release();
    return { success: true };
  } catch (err) {
    try { 
      await transaction.connection.rollback();
      transaction.connection.release();
    } catch {}
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: checkTimeout =====
function checkTimeout(ctx: Express.Request['ctx']): boolean {
  if (!ctx.deadline) return false;
  const isTimeout = Date.now() > ctx.deadline;
  if (isTimeout) {
    ctx.temp.success = false;
    ctx.temp.fail_reason = 'Request timeout';
  }
  return isTimeout;
}

// ===== 标准函数: Sha256 =====
async function Sha256(input: string): Promise<StandardResult<string>> {
  try {
    const salt = process.env.PASSWORD_SALT || '';
    if (!salt && process.env.NODE_ENV === 'production') {
      await LogError('警告：生产环境未配置PASSWORD_SALT', 'system', 'security', 'ERROR', 'system', '127.0.0.1');
    }
    const hash = createHash('sha256').update(salt + input).digest('hex');
    return { success: true, data: hash };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: ValidateParams =====
async function ValidateParams(params: Record<string, any>, schema: Joi.Schema): Promise<StandardResult<{ validated: RegisterRequest; errors: string[] }>> {
  try {
    const { error, value } = schema.validate(params, { abortEarly: false });
    if (error) {
      const errors = error.details.map(d => d.message);
      return { success: false, data: { errors } as any };
    }
    return { success: true, data: { validated: value as RegisterRequest, errors: [] } };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: IdempotentCheck =====
async function IdempotentCheck(idempotentKey: string, ttl: number, autoCache: boolean): Promise<StandardResult<{ is_first: boolean; cachedResponse?: any }>> {
  try {
    const exists = await redis.get(idempotentKey);
    if (exists) {
      try {
        const cachedData = JSON.parse(exists);
        return { success: true, data: { is_first: false, cachedResponse: cachedData } };
      } catch {
        return { success: true, data: { is_first: false, cachedResponse: { status: 'SUCCESS' } } };
      }
    }
    if (autoCache) {
      await redis.set(idempotentKey, JSON.stringify({ status: 'PROCESSING' }), 'EX', ttl);
    }
    return { success: true, data: { is_first: true } };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: IdempotentConfirm =====
async function IdempotentConfirm(idempotentKey: string, responseData: any): Promise<StandardResult> {
  try {
    await redis.set(idempotentKey, JSON.stringify(responseData), 'EX', CACHE_TTL.LONG);
    return { success: true };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
}

    return { success: true, data: str };

  }
}

// ===== 标准函数: DbQuery =====
async function DbQuery(tableName: string, conditions: Record<string, any>, options?: { transaction?: Transaction }): Promise<StandardResult<any[]>> {
  let connection: PoolConnection | null = null;
  try {
    if (options?.transaction) {
      connection = options.transaction.connection;
    } else {
      connection = await mysqlPool.getConnection();
    }
    const keys = Object.keys(conditions);
    const whereClause = keys.length ? `WHERE ${keys.map(k => `\`${k}\` = ?`).join(' AND ')}` : '';
    const values = keys.map(k => conditions[k]);
    const [rows] = await connection.query(`SELECT * FROM \`${tableName}\` ${whereClause}`, values);
    if (!options?.transaction) {
      connection.release();
    }
    return { success: true, data: rows as any[] };
  } catch (err) {
    if (connection && !options?.transaction) {
      try { connection.release(); } catch {}
    }
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: DbExecute =====
async function DbExecute(tableName: string, operation: 'insert' | 'update' | 'delete', data: Record<string, any>, conditions?: Record<string, any>, options?: { transaction?: Transaction }): Promise<StandardResult<ResultSetHeader>> {
  let connection: PoolConnection | null = null;
  try {
    if (options?.transaction) {
      connection = options.transaction.connection;
    } else {
      connection = await mysqlPool.getConnection();
    }
    let sql: string;
    let values: any[];
    if (operation === 'insert') {
      const keys = Object.keys(data);
      sql = `INSERT INTO \`${tableName}\` (${keys.map(k => `\`${k}\``).join(', ')}) VALUES (${keys.map(() => '?').join(', ')})`;
      values = keys.map(k => data[k]);
    } else if (operation === 'update') {
      const setKeys = Object.keys(data);
      const whereKeys = Object.keys(conditions || {});
      const setClause = setKeys.map(k => `\`${k}\` = ?`).join(', ');
      const whereClause = whereKeys.length ? `WHERE ${whereKeys.map(k => `\`${k}\` = ?`).join(' AND ')}` : '';
      sql = `UPDATE \`${tableName}\` SET ${setClause} ${whereClause}`;
      values = [...setKeys.map(k => data[k]), ...whereKeys.map(k => (conditions as Record<string, any>)[k])];
    } else { // delete
      const whereKeys = Object.keys(conditions || {});
      const whereClause = whereKeys.length ? `WHERE ${whereKeys.map(k => `\`${k}\` = ?`).join(' AND ')}` : '';
      sql = `DELETE FROM \`${tableName}\` ${whereClause}`;
      values = whereKeys.map(k => (conditions as Record<string, any>)[k]);
    }
    const [result] = await connection.query(sql, values);
    if (!options?.transaction) {
      connection.release();
    }
    return { success: true, data: result as ResultSetHeader };
  } catch (err) {
    if (connection && !options?.transaction) {
      try { connection.release(); } catch {}
    }
    return { success: false, error: (err as Error).message };
  }
}

// ===== 标准函数: LockAcquire =====
async function LockAcquire(lockKey: string, expireSeconds: number, retryTimes: number, isBlocking: boolean): Promise<StandardResult<{ locked: boolean; lockId?: string }>> {
  const lockId = `lock_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
  let attempts = 0;
  while (attempts <= retryTimes) {
    try {
      const result = await redis.set(lockKey, lockId, 'NX', 'EX', expireSeconds);
      if (result === 'OK') {
        return { success: true, data: { locked: true, lockId } };
      }
      if (!isBlocking) break;
      await new Promise(resolve => setTimeout(resolve, 100));
      attempts++;
    } catch (err) {
      return { success: false, error: (err as Error).message };
    }
  }
  return { success: true, data: { locked: false } };
}

// ===== 标准函数: LockRelease =====
async function LockRelease(lockKey: string, lockId: string): Promise<StandardResult> {
  try {
    // Lua脚本原子性校验lockId，防止误删其他进程的锁
    const script = `
      if redis.call('GET', KEYS[1]) == ARGV[1] then
        return redis.call('DEL', KEYS[1])
      else
        return 0
      end
    `;
    await redis.eval(script, 1, lockKey, lockId);
    return { success: true };
  } catch (err) {
    return { success: false, error: (err as Error).message };
  }
