import express, { Request, Response, NextFunction, Router } from 'express';
const router: Router = express.Router();
import {
    DbQuery, DbExecute, BeginTransaction, CommitTransaction, RollbackTransaction,
    LockAcquire, LockRelease, Log, LogError, checkTimeout,
    Sha256, ValidateParams, IdempotentCheck, IdempotentConfirm, SafeJsonStringify
} from '{basePath}/stdlib';
import { ERROR_CODE, USER_STATUS, CACHE_TTL } from '{basePath}/constants';

// 入参类型定义
interface RegisterRequest {
    username: string;
    password: string;
    email: string;
}

// 注册用户参数校验规则
const REGISTER_VALIDATION_RULES = {
    username: { required: true, type: 'string', min: 3, max: 32, pattern: '^[a-zA-Z0-9_-]+$' },
    password: { required: true, type: 'string', min: 8, max: 64 },
    email: { required: true, type: 'string', pattern: '^[^\\s@]+@[^\\s@]+\\.[^\\s@]+$' }
} as const;

router.post('/api/v1/user/register',
    async (req: Request, res: Response, next: NextFunction): Promise<void> => {
        const ctx = req.ctx;
        let transaction: { id: string } | null = null;
        let lockId: string | null = null;
        const LIGHT = false; // 第3站产出：第4位=1，属于标准级写入场景
        let registeredUserId: string | null = null;

        // ===== Stage 1: 初始化上下文 =====
        ctx.temp = ctx.temp || {};
        ctx.temp.success = true;
        ctx.temp.fail_reason = null;
        ctx.temp.logInfo = null;
        ctx.shared = ctx.shared || {};

        try {
            // ===== Stage 2: 前置校验与合规检查 =====
            // 请求体合法性校验
            const body = req.body as unknown as Record<string, unknown>;
            if (!body || typeof body !== 'object') {
                res.status(400).json({
                    code: ERROR_CODE.PARAM_INVALID,
                    msg: '请求体不能为空',
                    data: null,
                    detail: null
                });
                return;
            }

            // 结构化参数校验
            const validateResult = await ValidateParams(body, REGISTER_VALIDATION_RULES);
            if (!validateResult.success) {
                const errors = validateResult.data?.errors as string[] || [];
                res.status(400).json({
                    code: ERROR_CODE.PARAM_INVALID,
                    msg: `参数校验失败: ${errors.join(', ')}`,
                    data: null,
                    detail: null
                });
                return;
            }
            const validated = validateResult.data?.validated as Record<string, unknown>;
            const username = validated.username as string;
            const password = validated.password as string;
            const email = validated.email as string;

            // 幂等检查
            const idempotentKey = `user:register:${ctx.requestId}`;
            ctx.shared.idempotentKey = idempotentKey;
            const idemResult = await IdempotentCheck(idempotentKey, CACHE_TTL.LONG, true);
            if (!idemResult.success) {
                res.status(500).json({
                    code: ERROR_CODE.INTERNAL_ERROR,
                    msg: '幂等检查失败',
                    data: null,
                    detail: null
                });
                return;
            }
            if (!idemResult.data?.is_first) {
                const cached = idemResult.data?.cachedResponse || { username, status: 'SUCCESS' };
                res.status(200).json({
                    code: ERROR_CODE.SUCCESS,
                    msg: '注册请求已处理',
                    data: cached,
                    detail: null
                });
                return;
            }

            // 超时检查
            checkTimeout(ctx);
            if (ctx.temp.success === false) {
                throw new Error('TIMEOUT');
            }

            // ===== Stage 3: 核心业务执行 =====
            // 获取分布式锁（防止并发重复注册）
            const lockKey = `user:register:lock:${username}`;
            const lockResult = await LockAcquire(lockKey, 10, 3000, false);
            if (!lockResult.success || !lockResult.data?.locked) {
                res.status(429).json({
                    code: ERROR_CODE.LOCK_FAIL,
                    msg: '操作过于频繁，请稍后重试',
                    data: null,
                    detail: null
                });
                return;
            }
            lockId = lockResult.data.lockId;
            ctx.shared.lockKey = lockKey;
            ctx.shared.lockId = lockId;

            // 超时检查
            checkTimeout(ctx);
            if (ctx.temp.success === false) {
                throw new Error('TIMEOUT');
            }

            // 检查用户名是否已存在
            const usernameExistResult = await DbQuery('users', { username });
            if (!usernameExistResult.success) {
                ctx.temp.success = false;
                ctx.temp.fail_reason = usernameExistResult.error;
                throw new Error('DB_QUERY_FAILED');
            }
            if (usernameExistResult.data?.length > 0) {
                res.status(409).json({
                    code: ERROR_CODE.USER_ALREADY_EXISTS,
                    msg: '用户名已存在',
                    data: null,
                    detail: null
                });
                return;
            }

            // 检查邮箱是否已存在
            const emailExistResult = await DbQuery('users', { email });
            if (!emailExistResult.success) {
                ctx.temp.success = false;
                ctx.temp.fail_reason = emailExistResult.error;
                throw new Error('DB_QUERY_FAILED');
            }
            if (emailExistResult.data?.length > 0) {
                res.status(409).json({
                    code: ERROR_CODE.EMAIL_ALREADY_EXISTS,
                    msg: '邮箱已存在',
                    data: null,
                    detail: null
                });
                return;
            }

            // 密码哈希处理
            const passwordHashResult = await Sha256(password);
            if (!passwordHashResult.success) {
                ctx.temp.success = false;
                ctx.temp.fail_reason = passwordHashResult.error;
                throw new Error('PASSWORD_HASH_FAILED');
            }
            const passwordHash = passwordHashResult.data;

            // 开启事务
            checkTimeout(ctx);
            if (ctx.temp.success === false) {
                throw new Error('TIMEOUT');
            }
            const txResult = await BeginTransaction();
            if (!txResult.success) {
                ctx.temp.success = false;
                ctx.temp.fail_reason = txResult.error;
                throw new Error('TX_BEGIN_FAILED');
            }
            transaction = txResult.data;
            ctx.shared.transaction = transaction;

            // 写入用户数据
            const createUserResult = await DbExecute('users', 'insert', {
                username,
                password_hash: passwordHash,
                email,
                status: USER_STATUS.ACTIVE,
                created_at: new Date(),
                updated_at: new Date()
            }, undefined, { transaction });
            if (!createUserResult.success) {
                ctx.temp.success = false;
                ctx.temp.fail_reason = createUserResult.error;
                throw new Error('USER_CREATE_FAILED');
            }
            registeredUserId = (createUserResult.data as { insertId: string }).insertId;

            // ===== Stage 4: 后置归集与资源收尾 =====
            // 提交事务
            checkTimeout(ctx);
            if (ctx.temp.success === false) {
                throw new Error('TIMEOUT');
            }
            const commitResult = await CommitTransaction(transaction);
            if (!commitResult.success) {
                ctx.temp.success = false;
                ctx.temp.fail_reason = commitResult.error;
                throw new Error('COMMIT_FAILED');
            }
            transaction = null;

            // 释放锁
            if (lockId) {
                await LockRelease(lockKey, lockId);
                lockId = null;
            }

            // 幂等确认
            await IdempotentConfirm(idempotentKey);

            // 记录操作日志
            ctx.temp.logInfo = {
                action: 'user_register',
                username,
                email,
                userId: registeredUserId
            };
            const logUserId = ctx.userId || registeredUserId;
            await Log(ctx.temp.logInfo, logUserId, 'user', 'INFO', ctx.requestId, ctx.ip);

            // ===== Stage 5: 统一契约响应 =====
            const responseData = {
                userId: registeredUserId,
                username,
                email,
                status: USER_STATUS.ACTIVE
            };
            const jsonResult = await SafeJsonStringify(responseData, false);
            if (!jsonResult.success) {
                await LogError('JSON序列化失败: ' + jsonResult.error, logUserId, 'response', 'ERROR', ctx.requestId, ctx.ip);
            }

            res.status(200).json({
                code: ERROR_CODE.SUCCESS,
                msg: '用户注册成功',
                data: responseData,
                detail: null
            });
        } catch (err) {
            ctx.temp.success = false;
            if (!ctx.temp.fail_reason) {
                ctx.temp.fail_reason = (err as Error).message;
            }

            // 异常资源清理
            if (!LIGHT) {
                const lockKey = ctx.shared.lockKey as string;
                if (transaction) {
                    await RollbackTransaction(transaction);
                    transaction = null;
                }
                if (lockId) {
                    await LockRelease(lockKey, lockId);
                    lockId = null;
                }
            }

            // 错误日志记录
            const logUserId = ctx.userId || ctx.temp.loginUserId;
            await LogError(ctx.temp.fail_reason, logUserId, 'user_register', 'ERROR', ctx.requestId, ctx.ip);

            // 错误响应处理
            let errorCode: string = ERROR_CODE.INTERNAL_ERROR;
            let statusCode: number = 500;
            const errMsg = (err as Error).message;

            if (errMsg === 'TIMEOUT') {
                errorCode = ERROR_CODE.TIMEOUT;
                statusCode = 504;
            } else if (errMsg === 'DB_QUERY_FAILED') {
                errorCode = ERROR_CODE.INTERNAL_ERROR;
                statusCode = 500;
            } else if (errMsg === 'USER_CREATE_FAILED') {
                errorCode = ERROR_CODE.INTERNAL_ERROR;
                statusCode = 500;
            }

            res.status(statusCode).json({
                code: errorCode,
                msg: '注册失败，请稍后重试',
                data: null,
                detail: process.env.NODE_ENV === 'development' ? ctx.temp.fail_reason : null
            });
        }
    }
);

module.exports = router;
